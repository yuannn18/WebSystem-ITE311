# Git Setup Commands - Run in order after installing Git

# 1. Initialize Git repository
git init

# 2. Add all files
git add .

# 3. Make initial commit
git commit -m "Initial commit: CodeIgniter 4 with Bootstrap integration"

# 4. Add your GitHub repository (replace with your repo URL)
git remote add origin https://github.com/YOUR_USERNAME/YOUR_REPO_NAME.git

# 5. Push to GitHub
git push -u origin main

# Note: Make sure to create the repository on GitHub first!
